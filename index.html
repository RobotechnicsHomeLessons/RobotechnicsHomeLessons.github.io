2Бегущий огонь

Светодиоды загораются по очереди: 1 → 2 → 3. Одновременно горит только один.

Подсказка: прежде чем включить следующий, выключите предыдущий. Вспомните «Светофор».

3Шкала

Сначала горит 1, потом 1 и 2, потом 1, 2 и 3. Повторите это 2 раза, а в конце шкала «уменьшается»: 1+2+3 → 1+2 → 1.

Код 1: три светодиода

tri_svetodioda.ino
// ДОМАШНЕЕ ЗАДАНИЕ. Код 1: три светодиода
// Сейчас все три светодиода мигают ОДНОВРЕМЕННО.
// Ваша задача: менять только то, что внутри loop().

// --- Названия придумали МЫ сами. Можно назвать как угодно. ---
// Здесь запоминаем, к какому пину подключён каждый светодиод.
int led1 = 3;   // первый светодиод на пине 3
int led2 = 6;   // второй светодиод на пине 6
int led3 = 9;   // третий светодиод на пине 9

// setup() выполняется ОДИН раз при включении платы.
// setup, loop, pinMode, digitalWrite, delay - это СИСТЕМНЫЕ слова, их менять нельзя.
void setup() {
  pinMode(led1, OUTPUT);   // пин 3 будет ВЫДАВАТЬ сигнал
  pinMode(led2, OUTPUT);   // пин 6 будет ВЫДАВАТЬ сигнал
  pinMode(led3, OUTPUT);   // пин 9 будет ВЫДАВАТЬ сигнал
}

// loop() повторяется по кругу бесконечно.
void loop() {
  // digitalWrite(КУДА, ЧТО СДЕЛАТЬ)
  // HIGH - включить, LOW - выключить

  // Включаем все три
  digitalWrite(led1, HIGH);
  digitalWrite(led2, HIGH);
  digitalWrite(led3, HIGH);
  delay(500);   // ждём 500 миллисекунд (полсекунды)

  // Выключаем все три
  digitalWrite(led1, LOW);
  digitalWrite(led2, LOW);
  digitalWrite(led3, LOW);
  delay(500);   // ждём полсекунды
}

Код 2: датчик света

datchik_sveta.ino
// ДОМАШНЕЕ ЗАДАНИЕ. Код 2: датчик света
// Плата читает датчик и пишет число на экран компьютера.
// Цифровое значение - это только 0 или 1: свет есть или света нет.

// Здесь запоминаем, к какому пину подключён датчик.
// Переподключили датчик на другой пин - поменяйте число здесь!
int sensorPin = 2;

void setup() {
  pinMode(sensorPin, INPUT);   // пин будет ПРИНИМАТЬ сигнал (вход)
  Serial.begin(9600);          // включаем связь с компьютером
}

void loop() {
  // digitalRead(ОТКУДА) - прочитать, что сейчас на пине: 0 или 1
  int value = digitalRead(sensorPin);

  Serial.println(value);   // написать число на экран
  delay(200);              // ждём 0,2 секунды и читаем снова
}

// Как посмотреть числа:
// 1. Загрузите код на плату.
// 2. Нажмите на значок лупы в правом верхнем углу Arduino IDE
//    (Инструменты -> Монитор порта).
// 3. Внизу окна выберите скорость 9600.

Дополнительные программы

Примеры для экспериментов с цифровыми выходами, входами и кнопкой или датчиком. Каждый фрагмент — отдельная программа: загружайте их по одной.

Программа 1: включить выход 10
void setup() {
  pinMode(10, OUTPUT);
}

void loop() {
  digitalWrite(10, HIGH);
}
Программа 2: повторить сигнал с пина 2 на пин 13
void setup() {
  pinMode(13, OUTPUT);
  pinMode(2, INPUT);
}

void loop() {
  digitalWrite(13, digitalRead(2));
}
Программа 3: включить пин 13 при сигнале на пине 2
void setup() {
  pinMode(13, OUTPUT);
  pinMode(2, INPUT);
}

void loop() {
  if (digitalRead(2) == 1) {
    digitalWrite(13, HIGH);
  }
}
Программа 4: мигание светодиодом
void setup() {
  pinMode(13, OUTPUT);  // настраиваем пин 13 как выход
}

void loop() {
  digitalWrite(13, HIGH);  // включаем светодиод
  delay(700);              // ждём 700 мс

  digitalWrite(13, LOW);   // выключаем светодиод
  delay(300);              // ждём 300 мс

  // Цикл повторяется: полный период мигания — 1 секунда
}
Программа 5: два выхода в зависимости от входа
void setup() {
  pinMode(13, OUTPUT);
  pinMode(12, OUTPUT);
  pinMode(2, INPUT);
}

void loop() {
  if (digitalRead(2) == HIGH) {
    digitalWrite(13, HIGH);
    digitalWrite(12, LOW);
  } else {
    digitalWrite(12, HIGH);
    digitalWrite(13, LOW);
  }
}
Программа 6: переключение состояния по сигналу
bool state = true;

void setup() {
  pinMode(13, OUTPUT);
  pinMode(2, INPUT);
}

void loop() {
  if (digitalRead(2) == HIGH) {
    delay(300);
    state = !state;
  }

  digitalWrite(13, state);
}
Важно: программы загружаются по одной. Для входа на пине 2 используйте корректную схему подключения и не оставляйте вход «в воздухе» — иначе его состояние может считываться нестабильно.
